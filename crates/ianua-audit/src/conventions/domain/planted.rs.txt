impl PlantedRecord {
    // ── Planted indented section ──
    // Planted — an em dash.

    pub fn reconstitute(owner_id: Uuid) -> Self {
        Self::from(owner_id)
    }

    pub fn transition_to(self) {}

    pub fn parse(raw: &str) -> Result<Uuid, DomainError> {
        if raw.is_empty() {
            return Err(DomainError::Validation(ValidationError::Rule {
                message: "empty".to_owned(),
            }));
        }
        Err(DomainError::infra_msg(raw))
    }

    #[allow(dead_code)]
    pub fn close(&mut self, early: bool, forced: bool) -> Uuid {
        // closing early skips the grace period
        if early {
            if forced {
                self.status = PlantedStatus::Closed;
            }
        }
        Uuid::now_v7()
    }
}

pub fn planted_repeat(values: &[&str]) -> bool {
    let mut seen = HashSet::new();
    values.iter().any(|value| !seen.insert(value))
}

pub fn planted_option(limit: Option<u32>) -> u32 {
    limit.unwrap_or(0)
}
